
#ifndef PWM_INPUT_MANAGER_HPP
#define PWM_INPUT_MANAGER_HPP

#include "main.h"
#include "PinMap.h"
#include <stdint.h>

class PWMInputManager
{
public:
    static bool Init();

    // Input channel numbers: 1 to 8
    static bool Start(uint8_t channel);
    static bool Stop(uint8_t channel);

    static bool StartAll();
    static bool StopAll();

    // Measured PWM values
    static uint32_t GetFrequency(uint8_t channel);
    static float GetDuty(uint8_t channel);
    static uint32_t GetPeriodUs(uint8_t channel);
    static uint32_t GetPulseWidthUs(uint8_t channel);

    static bool IsRunning(uint8_t channel);
    static bool IsValid(uint8_t channel);

    // Forward HAL input-capture callbacks here.
    static void HandleCapture(TIM_HandleTypeDef* htim);

private:
    struct CaptureState
    {
        volatile uint32_t lastRising;
        volatile uint32_t highTicks;

        volatile uint32_t frequencyHz;
        volatile uint32_t periodUs;
        volatile uint32_t pulseWidthUs;
        volatile float dutyPercent;

        volatile bool haveRising;
        volatile bool haveHigh;
        volatile bool waitingForFalling;
        volatile bool running;
        volatile bool valid;
    };

    static CaptureState state[8];

    static bool IsValidChannel(uint8_t channel);

    static uint32_t GetTimerClock(TIM_HandleTypeDef* timer);
    static uint32_t GetCounterClock(TIM_HandleTypeDef* timer);

    static uint32_t GetElapsedTicks(
        TIM_HandleTypeDef* timer,
        uint32_t start,
        uint32_t end
    );

    static uint32_t GetTimerChannel(uint32_t activeChannel);
};

#endif
